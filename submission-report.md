1. From the window 1, no loss run, compute the round trip time your sender actually saw. (It sent 1025 packets and waited one round trip for each.) The relay adds 100 ms. Where does the rest come from?

For a windows size of 1 with no loss, there was an observed mean RTT of 100.785ms. 
100ms of the RTT came from the relay, while additional delay was created while the
sender and reciver were reading, validating, and generating responses to incoming packets. 

2. Use that round trip time to explain the speedup at window 16. Is it close to 16 times? Why or why not?

For a windows size of 16 with no loss, there was an observed mean RTT of 6.482ms.
With the reduction of RTT to almost 1/16 when compared to the windows size 1, no loss run,
the transfer time was also reduced because the average time taken for any given packet to 
be send and acked was redueced.

```c
static int parse_unsigned(const char *text, unsigned long maximum,
                          unsigned long *value)
{
  char *end = NULL;
  unsigned long parsed;

  if (text == NULL || text[0] == '\0' || text[0] == '-')
  {
    return -1;
  }

  errno = 0;
  parsed = strtoul(text, &end, 10);
  if (errno != 0 || end == text || *end != '\0' || parsed > maximum)
  {
    return -1;
  }

  *value = parsed;
  return 0;
}

static int parse_probability(const char *text, double *value)
{
  char *end = NULL;
  double parsed;

  if (text == NULL || text[0] == '\0')
  {
    return -1;
  }

  errno = 0;
  parsed = strtod(text, &end);
  if (errno != 0 || end == text || *end != '\0' || !isfinite(parsed) ||
      parsed < 0.0 || parsed > 0.5)
  {
    return -1;
  }

  *value = parsed;
  return 0;
}

static int parse_send(int argc, char **argv)
{
  struct rdt_options options = {
    .window_size = DEFAULT_WINDOW,
    .timeout_ms = DEFAULT_TIMEOUT_MS,
    .port = DEFAULT_PORT,
    .loss = 0.0,
    .corrupt = 0.0,
    .duplicate = 0.0
  };
  int option;
  int result;

  optind = 2;
  while ((option = getopt(argc, argv, "s:w:T:l:c:d:p:")) != -1)
  {
    unsigned long parsed;
    switch (option)
    {
      case 's':
        options.session = optarg;
        break;
      case 'w':
        if (parse_unsigned(optarg, 64UL, &parsed) != 0 || parsed == 0UL)
        {
          fprintf(stderr, "Invalid window size: %s\n", optarg);
          print_usage(stderr);
          return 2;
        }
        options.window_size = (unsigned int) parsed;
        break;
      case 'T':
        if (parse_unsigned(optarg, MAX_TIMEOUT_MS, &parsed) != 0 ||
            parsed == 0UL)
        {
          fprintf(stderr, "Invalid timeout: %s (expected 1-%u ms)\n",
                  optarg, MAX_TIMEOUT_MS);
          print_usage(stderr);
          return 2;
        }
        options.timeout_ms = (unsigned int) parsed;
        break;
      case 'l':
        if (parse_probability(optarg, &options.loss) != 0)
        {
          fprintf(stderr, "Invalid loss probability: %s\n", optarg);
          print_usage(stderr);
          return 2;
        }
        break;
      case 'c':
        if (parse_probability(optarg, &options.corrupt) != 0)
        {
          fprintf(stderr, "Invalid corruption probability: %s\n", optarg);
          print_usage(stderr);
          return 2;
        }
        break;
      case 'd':
        if (parse_probability(optarg, &options.duplicate) != 0)
        {
          fprintf(stderr, "Invalid duplication probability: %s\n", optarg);
          print_usage(stderr);
          return 2;
        }
        break;
      case 'p':
        if (parse_unsigned(optarg, 65535UL, &parsed) != 0 || parsed == 0UL)
        {
          fprintf(stderr, "Invalid relay port: %s\n", optarg);
          print_usage(stderr);
          return 2;
        }
        options.port = (unsigned short) parsed;
        break;
      default:
        print_usage(stderr);
        return 2;
    }
  }

  if (options.session == NULL || optind + 2 != argc)
  {
    print_usage(stderr);
    return 2;
  }
  result = rdt_validate_session(options.session);
  if (result != 0)
  {
    fprintf(stderr,
            "Invalid session: use 1-32 characters from a-z, 0-9, and '-'\n");
    print_usage(stderr);
    return 2;
  }

  return rdt_send_file(argv[optind], options.port, argv[optind + 1], &options);
}

static int parse_receive(int argc, char **argv)
{
  struct rdt_options options = {
    .window_size = DEFAULT_WINDOW,
    .timeout_ms = DEFAULT_TIMEOUT_MS,
    .port = DEFAULT_PORT,
    .loss = 0.0,
    .corrupt = 0.0,
    .duplicate = 0.0
  };
  int option;

  optind = 2;
  while ((option = getopt(argc, argv, "s:p:")) != -1)
  {
    unsigned long parsed;
    switch (option)
    {
      case 's':
        options.session = optarg;
        break;
      case 'p':
        if (parse_unsigned(optarg, 65535UL, &parsed) != 0 || parsed == 0UL)
        {
          fprintf(stderr, "Invalid relay port: %s\n", optarg);
          print_usage(stderr);
          return 2;
        }
        options.port = (unsigned short) parsed;
        break;
      default:
        print_usage(stderr);
        return 2;
    }
  }

  if (options.session == NULL || optind + 2 != argc)
  {
    print_usage(stderr);
    return 2;
  }
  if (rdt_validate_session(options.session) != 0)
  {
    fprintf(stderr,
            "Invalid session: use 1-32 characters from a-z, 0-9, and '-'\n");
    print_usage(stderr);
    return 2;
  }

  return rdt_receive_file(argv[optind], options.port, argv[optind + 1],
                          &options);
}

int main(int argc, char **argv)
{
  if (argc < 2)
  {
    print_usage(stderr);
    return 0;
  }
  if (strcmp(argv[1], "send") == 0)
  {
    return parse_send(argc, argv);
  }
  if (strcmp(argv[1], "recv") == 0)
  {
    return parse_receive(argc, argv);
  }

  print_usage(stderr);
  return 2;
}
```

```c
#include "rdt.h"

#include <arpa/inet.h>
#include <string.h>

/*
 * Packet serialization layer. It owns the wire format and does not perform
 * network, timer, or file operations.
 */
static uint16_t read_u16(const uint8_t *bytes)
{
  uint16_t network_value;
  memcpy(&network_value, bytes, sizeof(network_value));
  return ntohs(network_value);
}

static uint32_t read_u32(const uint8_t *bytes)
{
  uint32_t network_value;
  memcpy(&network_value, bytes, sizeof(network_value));
  return ntohl(network_value);
}

static void write_u16(uint8_t *bytes, uint16_t value)
{
  uint16_t network_value = htons(value);
  memcpy(bytes, &network_value, sizeof(network_value));
}

static void write_u32(uint8_t *bytes, uint32_t value)
{
  uint32_t network_value = htonl(value);
  memcpy(bytes, &network_value, sizeof(network_value));
}

static uint16_t checksum_bytes(const uint8_t *bytes, size_t length,
                               int zero_checksum_field)
{
  uint32_t sum = 0U;
  size_t offset = 0U;

  if (bytes == NULL && length > 0U)
  {
    return 0U;
  }
  while (offset + 1U < length)
  {
    uint16_t word = zero_checksum_field != 0 && offset == 2U
                        ? 0U
                        : (uint16_t) (((uint16_t) bytes[offset] << 8) |
                                      (uint16_t) bytes[offset + 1U]);
    sum += (uint32_t) word;
    offset += 2U;
  }
  if (offset < length)
  {
    sum += (uint32_t) ((uint16_t) bytes[offset] << 8);
  }
  while ((sum >> 16) != 0U)
  {
    sum = (sum & UINT32_C(0xffff)) + (sum >> 16);
  }
  return (uint16_t) ~sum;
}

uint16_t rdt_checksum(const uint8_t *bytes, size_t length)
{
  return checksum_bytes(bytes, length, 0);
}

int rdt_validate_session(const char *session)
{
  size_t i;
  size_t length;

  if (session == NULL)
  {
    return -1;
  }
  length = strlen(session);
  if (length == 0U || length > 32U)
  {
    return -1;
  }
  for (i = 0U; i < length; ++i)
  {
    char character = session[i];
    if (!((character >= 'a' && character <= 'z') ||
          (character >= '0' && character <= '9') || character == '-'))
    {
      return -1;
    }
  }
  return 0;
}

size_t rdt_encode_packet(uint8_t *packet, size_t capacity,
                         enum rdt_packet_type type, uint32_t sequence,
                         const uint8_t *payload, uint32_t payload_length)
{
  size_t packet_length;

  if (packet == NULL || payload_length > RDT_MAX_PAYLOAD ||
      (payload_length > 0U && payload == NULL) ||
      (type != RDT_DATA && type != RDT_ACK && type != RDT_FIN))
  {
    return 0U;
  }
  packet_length = RDT_HEADER_SIZE + (size_t) payload_length;
  if (capacity < packet_length)
  {
    return 0U;
  }

  packet[0] = (uint8_t) type;
  packet[1] = 0U;
  write_u16(packet + 2U, 0U);
  write_u32(packet + 4U, sequence);
  write_u16(packet + 8U, (uint16_t) payload_length);
  if (payload_length > 0U)
  {
    memcpy(packet + RDT_HEADER_SIZE, payload, (size_t) payload_length);
  }
  write_u16(packet + 2U, checksum_bytes(packet, packet_length, 1));
  return packet_length;
}

int rdt_decode_packet(const uint8_t *packet, size_t packet_length,
                      enum rdt_packet_type *type, uint32_t *sequence,
                      const uint8_t **payload, uint32_t *payload_length)
{
  uint16_t decoded_length;

  if (packet == NULL || packet_length < RDT_HEADER_SIZE || packet[1] != 0U ||
      (packet[0] != RDT_DATA && packet[0] != RDT_ACK &&
       packet[0] != RDT_FIN))
  {
    return -1;
  }

  decoded_length = read_u16(packet + 8U);
  if (decoded_length > RDT_MAX_PAYLOAD ||
      packet_length != RDT_HEADER_SIZE + (size_t) decoded_length ||
      rdt_checksum(packet, packet_length) != 0U)
  {
    return -1;
  }

  if (type != NULL)
  {
    *type = (enum rdt_packet_type) packet[0];
  }
  if (sequence != NULL)
  {
    *sequence = read_u32(packet + 4U);
  }
  if (payload != NULL)
  {
    *payload = packet + RDT_HEADER_SIZE;
  }
  if (payload_length != NULL)
  {
    *payload_length = decoded_length;
  }
  return 0;
}
```

### rdt.h

```c

#ifndef RDT_H
#define RDT_H

#include <stddef.h>
#include <stdint.h>

/** Size in bytes of the fixed portion of an RDT datagram. */
#define RDT_HEADER_SIZE 10U

/** Maximum number of file-data bytes carried by one DATA datagram. */
#define RDT_MAX_PAYLOAD 1024U

/** Maximum total RDT datagram size, including its header. */
#define RDT_MAX_PACKET (RDT_HEADER_SIZE + RDT_MAX_PAYLOAD)

/**
 * RDT datagram type values. These values are part of the wire protocol.
 */
enum rdt_packet_type
{
  RDT_DATA = 0,
  RDT_ACK = 1,
  RDT_FIN = 2
};

/**
 * Configuration for a transfer and its relay registration.
 *
 * The session string is used during the relay HELLO handshake. window_size and
 * timeout_ms are used by the sender; loss, corrupt, and duplicate are sent to
 * the relay in the sender's HELLO. port selects the relay's UDP port.
 */
struct rdt_options
{
  const char *session;       /**< Shared relay session name. */
  unsigned int window_size;  /**< Sender's Go-Back-N window, 1-64 packets. */
  unsigned int timeout_ms;   /**< Sender retransmission timeout in ms. */
  unsigned short port;       /**< Relay UDP port. */
  double loss;               /**< Relay packet-loss probability, 0-0.5. */
  double corrupt;            /**< Relay packet-corruption probability, 0-0.5. */
  double duplicate;          /**< Relay duplication probability, 0-0.5. */
};

/**
 * Check the session-name syntax expected by the relay.
 *
 * @param session Name to validate; must contain 1-32 lowercase letters,
 *                digits, or hyphens.
 * @return 0 when valid, or -1 for NULL or an invalid name.
 */
int rdt_validate_session(const char *session);

/**
 * Compute the Internet checksum (RFC 1071) for a byte sequence.
 * An odd final byte is treated as the high byte of a zero-padded 16-bit word.
 * Returns zero for a NULL byte pointer with a nonzero length.
 */
uint16_t rdt_checksum(const uint8_t *bytes, size_t length);

/**
 * Encode one RDT datagram into its wire representation.
 *
 * The header consists of type, a zero reserved byte, a 16-bit Internet
 * checksum, a 32-bit sequence number, and a 16-bit payload length. Multi-byte
 * fields are in network byte order.
 *
 * @param packet Destination buffer.
 * @param capacity Available bytes in packet.
 * @param type DATA, ACK, or FIN.
 * @param sequence 32-bit datagram sequence number.
 * @param payload Payload bytes, or NULL when payload_length is zero.
 * @param payload_length Payload size, at most RDT_MAX_PAYLOAD.
 * @return Encoded datagram length, or 0 if the arguments/capacity are invalid.
 */
size_t rdt_encode_packet(uint8_t *packet, size_t capacity,
                         enum rdt_packet_type type, uint32_t sequence,
                         const uint8_t *payload, uint32_t payload_length);

/**
 * Validate and decode an RDT datagram.
 *
 * Optional output pointers may be NULL. The decoded payload points into the
 * input packet and remains valid only as long as that packet's storage.
 *
 * @param packet Datagram bytes to validate.
 * @param packet_length Datagram size.
 * @param type Receives the decoded type, when non-NULL.
 * @param sequence Receives the decoded sequence number, when non-NULL.
 * @param payload Receives a pointer to the payload, when non-NULL.
 * @param payload_length Receives the payload size, when non-NULL.
 * @return 0 on success, or -1 for invalid type, reserved byte, length, or
 *         checksum.
 */
int rdt_decode_packet(const uint8_t *packet, size_t packet_length,
                      enum rdt_packet_type *type, uint32_t *sequence,
                      const uint8_t **payload, uint32_t *payload_length);

/**
 * Send a regular file through the relay using Go-Back-N.
 *
 * Registers as the session sender, sends the file in RDT_MAX_PAYLOAD-sized
 * DATA chunks, retransmits unacknowledged windows on timeout, then retries FIN
 * until its final ACK arrives or ten timeouts without progress expire.
 *
 * @param host Relay host name or numeric address.
 * @param port Relay UDP port.
 * @param path Path to the regular input file.
 * @param options Transfer and relay settings.
 * @return 0 on success, 2 if DATA or FIN makes no progress after ten
 *         consecutive timeouts, or 1 for other failures.
 */
int rdt_send_file(const char *host, unsigned short port, const char *path,
                  const struct rdt_options *options);

/**
 * Receive one file through the relay.
 *
 * Registers as the session receiver, writes only DATA at the next expected
 * sequence number, sends cumulative ACKs for received or discarded packets,
 * then acknowledges FIN and lingers through the sender's bounded FIN retry
 * period.
 *
 * @param host Relay host name or numeric address.
 * @param port Relay UDP port.
 * @param path Destination file path; it is opened/truncated after registration.
 * @param options Transfer and relay settings.
 * @return 0 on success, 2 if no valid packet arrives for 30 seconds, or 1 for
 *         other file or communication failures.
 */
int rdt_receive_file(const char *host, unsigned short port, const char *path,
                     const struct rdt_options *options);

#endif
```

```c
#define _POSIX_C_SOURCE 200809L

#include "rdt_io.h"

#include "rdt.h"

#include <errno.h>
#include <inttypes.h>
#include <netdb.h>
#include <poll.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/socket.h>
#include <sys/stat.h>
#include <time.h>
#include <unistd.h>

#define HELLO_ATTEMPTS 5
#define HELLO_TIMEOUT_MS 1000

struct rdt_io_context
{
  int socket_fd;
  FILE *file;
  int sender;
};

static int open_relay_socket(const char *host, unsigned short port)
{
  struct addrinfo hints;
  struct addrinfo *addresses = NULL;
  struct addrinfo *address;
  char service[6];
  int socket_fd = -1;
  int status;

  memset(&hints, 0, sizeof(hints));
  hints.ai_family = AF_UNSPEC;
  hints.ai_socktype = SOCK_DGRAM;
  hints.ai_protocol = IPPROTO_UDP;
  (void) snprintf(service, sizeof(service), "%u", (unsigned int) port);

  status = getaddrinfo(host, service, &hints, &addresses);
  // GCOVR_EXCL_START
  if (status != 0)
  {
    fprintf(stderr, "Cannot resolve relay '%s': %s\n", host,
            gai_strerror(status));
    return -1;
  }
  // GCOVR_EXCL_STOP

  // GCOVR_EXCL_START
  for (address = addresses; address != NULL; address = address->ai_next)
  {
    socket_fd = socket(address->ai_family, address->ai_socktype,
                       address->ai_protocol);
    if (socket_fd < 0)
    {
      continue;
    }
    if (connect(socket_fd, address->ai_addr, address->ai_addrlen) == 0)
    {
      break;
    }
    (void) close(socket_fd);
    socket_fd = -1;
  }
  // GCOVR_EXCL_STOP
  freeaddrinfo(addresses);

  // GCOVR_EXCL_START
  if (socket_fd < 0)
  {
    fprintf(stderr, "Cannot connect UDP socket to relay '%s:%u': %s\n",
            host, (unsigned int) port, strerror(errno));
  }
  // GCOVR_EXCL_STOP
  return socket_fd;
}

static int send_datagram(void *context, const uint8_t *bytes, size_t length)
{
  struct rdt_io_context *io = context;
  ssize_t sent;

  // GCOVR_EXCL_START
  do
  {
    sent = send(io->socket_fd, bytes, length, 0);
  } while (sent < 0 && errno == EINTR);

  if (sent < 0 || (size_t) sent != length)
  {
    if (sent >= 0)
    {
      fprintf(stderr, "Short UDP datagram send\n");
    }
    else
    {
      fprintf(stderr, "UDP send failed: %s\n", strerror(errno));
    }
    return -1;
  }
  // GCOVR_EXCL_STOP
  return 0;
}

static int wait_readable(int socket_fd, int timeout_ms)
{
  struct pollfd descriptor = {
    .fd = socket_fd,
    .events = POLLIN,
    .revents = 0
  };
  int result;

  // GCOVR_EXCL_START
  do
  {
    result = poll(&descriptor, 1U, timeout_ms);
  } while (result < 0 && errno == EINTR);

  if (result < 0)
  {
    fprintf(stderr, "UDP wait failed: %s\n", strerror(errno));
    return -1;
  }
  // GCOVR_EXCL_STOP
  if (result == 0)
  {
    return 0;
  }
  // GCOVR_EXCL_START
  if ((descriptor.revents & POLLIN) != 0)
  {
    return 1;
  }
  fprintf(stderr, "UDP socket reported an error while waiting\n");
  return -1;
  // GCOVR_EXCL_STOP
}

static uint64_t monotonic_milliseconds(void *context)
{
  struct timespec now;
  (void) context;

  // GCOVR_EXCL_START
  if (clock_gettime(CLOCK_MONOTONIC, &now) != 0)
  {
    fprintf(stderr, "Could not read monotonic clock: %s\n", strerror(errno));
    return 0U;
  }
  // GCOVR_EXCL_STOP
  return (uint64_t) now.tv_sec * UINT64_C(1000) +
         (uint64_t) now.tv_nsec / UINT64_C(1000000);
}

static int read_file(void *context, uint8_t *bytes, size_t length)
{
  struct rdt_io_context *io = context;

  // GCOVR_EXCL_START
  if (fread(bytes, 1U, length, io->file) != length)
  {
    fprintf(stderr, "Could not read input file: %s\n",
            ferror(io->file) != 0 ? strerror(errno) : "unexpected end of file");
    return -1;
  }
  // GCOVR_EXCL_STOP
  return 0;
}

static int write_file(void *context, const uint8_t *bytes, size_t length)
{
  struct rdt_io_context *io = context;

  // GCOVR_EXCL_START
  if (fwrite(bytes, 1U, length, io->file) != length)
  {
    fprintf(stderr, "Could not write output file: %s\n", strerror(errno));
    return -1;
  }
  // GCOVR_EXCL_STOP
  return 0;
}

static int flush_file(void *context)
{
  struct rdt_io_context *io = context;

  // GCOVR_EXCL_START
  if (fflush(io->file) != 0)
  {
    fprintf(stderr, "Could not flush output file: %s\n", strerror(errno));
    return -1;
  }
  // GCOVR_EXCL_STOP
  return 0;
}

static int close_file(void *context)
{
  struct rdt_io_context *io = context;
  FILE *file = io->file;

  io->file = NULL;
  // GCOVR_EXCL_START
  if (fclose(file) != 0)
  {
    fprintf(stderr, "Could not close output file: %s\n", strerror(errno));
    return -1;
  }
  // GCOVR_EXCL_STOP
  return 0;
}

static int relay_handshake(struct rdt_io_context *io,
                           const struct rdt_options *options)
{
  char hello[128];
  int hello_length;
  int attempt;

  if (io->sender != 0)
  {
    hello_length = snprintf(hello, sizeof(hello),
                            "HELLO %s send %.17f %.17f %.17f",
                            options->session, options->loss,
                            options->corrupt, options->duplicate);
  }
  else
  {
    hello_length = snprintf(hello, sizeof(hello), "HELLO %s recv",
                            options->session);
  }
  // GCOVR_EXCL_START
  if (hello_length < 0 || (size_t) hello_length >= sizeof(hello))
  {
    fprintf(stderr, "Could not construct relay HELLO message\n");
    return -1;
  }
  // GCOVR_EXCL_STOP

  /* HELLO and its reply are not impaired; retries cover UDP loss. */
  for (attempt = 0; attempt < HELLO_ATTEMPTS; ++attempt)
  {
    uint8_t response[256];
    size_t response_length = 0U;
    int ready;
    ssize_t received;

    // GCOVR_EXCL_START
    if (send_datagram(io, (const uint8_t *) hello,
                      (size_t) hello_length) != 0)
    {
      return -1;
    }
    // GCOVR_EXCL_STOP
    ready = wait_readable(io->socket_fd, HELLO_TIMEOUT_MS);
    // GCOVR_EXCL_START
    if (ready < 0)
    {
      return -1;
    }
    // GCOVR_EXCL_STOP
    if (ready == 0)
    {
      continue;
    }
    // GCOVR_EXCL_START
    do
    {
      received = recv(io->socket_fd, response, sizeof(response) - 1U, 0);
    } while (received < 0 && errno == EINTR);
    if (received < 0)
    {
      fprintf(stderr, "UDP receive failed: %s\n", strerror(errno));
      return -1;
    }
    // GCOVR_EXCL_STOP
    response_length = (size_t) received;
    response[response_length] = '\0';
    if (response_length == 2U && memcmp(response, "OK", 2U) == 0)
    {
      return 0;
    }
    if (response_length >= 4U && memcmp(response, "ERR ", 4U) == 0)
    {
      fprintf(stderr, "Relay rejected HELLO: %s\n", response + 4U);
    }
    else
    {
      fprintf(stderr, "Unexpected relay HELLO response: %s\n", response);
    }
    return -1;
  }
  fprintf(stderr, "Relay did not answer HELLO after %d attempts\n",
          HELLO_ATTEMPTS);
  return -1;
}

struct rdt_io_context *rdt_io_open(const char *host, unsigned short port,
                                   const char *path,
                                   const struct rdt_options *options,
                                   int sender, uint64_t *file_size)
{
  struct rdt_io_context *io;
  struct stat file_status;

  if (host == NULL || path == NULL || options == NULL ||
      rdt_validate_session(options->session) != 0 ||
      (sender != 0 && sender != 1))
  {
    fprintf(stderr, "Invalid endpoint or relay options\n");
    return NULL;
  }

  io = calloc(1U, sizeof(*io));
  // GCOVR_EXCL_START
  if (io == NULL)
  {
    fprintf(stderr, "Could not allocate I/O context: %s\n", strerror(errno));
    return NULL;
  }
  // GCOVR_EXCL_STOP
  io->socket_fd = -1;
  io->sender = sender;

  if (sender != 0)
  {
    // GCOVR_EXCL_START
    io->file = fopen(path, "rb");
    if (io->file == NULL)
    {
      fprintf(stderr, "Cannot open input file '%s': %s\n", path,
              strerror(errno));
      rdt_io_close(io);
      return NULL;
    }
    // GCOVR_EXCL_STOP
    // GCOVR_EXCL_START
    if (fstat(fileno(io->file), &file_status) != 0)
    {
      fprintf(stderr, "Input must be a readable regular file\n");
      rdt_io_close(io);
      return NULL;
    }
    // GCOVR_EXCL_STOP
    if (!S_ISREG(file_status.st_mode))
    {
      fprintf(stderr, "Input must be a readable regular file\n");
      rdt_io_close(io);
      return NULL;
    }
    if (file_size != NULL)
    {
      *file_size = (uint64_t) file_status.st_size;
    }
  }

  io->socket_fd = open_relay_socket(host, port);
  // GCOVR_EXCL_START
  if (io->socket_fd < 0)
  {
    rdt_io_close(io);
    return NULL;
  }
  // GCOVR_EXCL_STOP
  if (relay_handshake(io, options) != 0)
  {
    rdt_io_close(io);
    return NULL;
  }

  if (sender == 0)
  {
    io->file = fopen(path, "wb");
    if (io->file == NULL)
    {
      fprintf(stderr, "Cannot open output file '%s': %s\n", path,
              strerror(errno));
      rdt_io_close(io);
      return NULL;
    }
  }
  return io;
}

int rdt_io_close(struct rdt_io_context *context)
{
  int result = 0;

  if (context == NULL)
  {
    return 0;
  }
  if (context->file != NULL)
  {
    // GCOVR_EXCL_START
    if (fclose(context->file) != 0)
    {
      fprintf(stderr, "Could not close transfer file: %s\n", strerror(errno));
      result = -1;
    }
    // GCOVR_EXCL_STOP
  }
  if (context->socket_fd >= 0)
  {
    // GCOVR_EXCL_START
    if (close(context->socket_fd) != 0)
    {
      fprintf(stderr, "Could not close relay socket: %s\n", strerror(errno));
      result = -1;
    }
    // GCOVR_EXCL_STOP
  }
  free(context);
  return result;
}

static int receive_protocol_packet(struct rdt_io_context *io,
                                   uint64_t deadline_ms,
                                   struct rdt_protocol_packet *packet)
{
  uint8_t bytes[RDT_MAX_PACKET];

  for (;;)
  {
    uint64_t now = monotonic_milliseconds(io);
    int timeout_ms;
    int ready;
    ssize_t received;
    const uint8_t *payload = NULL;

    if (now >= deadline_ms)
    {
      return 1;
    }
    timeout_ms = (int) (deadline_ms - now);
    ready = wait_readable(io->socket_fd, timeout_ms);
    if (ready == 0)
    {
      return 1;
    }
    // GCOVR_EXCL_START
    if (ready < 0)
    {
      return -1;
    }
    // GCOVR_EXCL_STOP
    // GCOVR_EXCL_START
    do
    {
      received = recv(io->socket_fd, bytes, sizeof(bytes), 0);
    } while (received < 0 && errno == EINTR);
    if (received < 0)
    {
      fprintf(stderr, "UDP receive failed: %s\n", strerror(errno));
      return -1;
    }
    // GCOVR_EXCL_STOP
    if (rdt_decode_packet(bytes, (size_t) received, &packet->type,
                          &packet->sequence, &payload,
                          &packet->payload_length) == 0)
    {
      if (packet->payload_length > 0U)
      {
        memcpy(packet->payload, payload, packet->payload_length);
      }
      return 0;
    }
  }
}

static int sender_dispatch(struct rdt_io_context *io,
                           struct rdt_sender *sender,
                           const struct rdt_protocol_actions *actions)
{
  struct rdt_protocol_actions current = *actions;
  size_t i;

process_batch:
  for (i = 0U; i < current.count; ++i)
  {
    const struct rdt_protocol_action *action = &current.items[i];
    struct rdt_protocol_actions next_actions;
    struct rdt_sender_event event;
    int result;

    switch (action->type)
    {
    case RDT_ACTION_READ_FILE:
    {
      uint8_t bytes[RDT_MAX_PAYLOAD];
      // GCOVR_EXCL_START
      if (read_file(io, bytes, action->length) != 0)
      {
        memset(&event, 0, sizeof(event));
        event.type = RDT_SENDER_EVENT_IO_FAILURE;
      }
      else
      {
        memset(&event, 0, sizeof(event));
        event.type = RDT_SENDER_EVENT_DATA_READY;
        event.now_ms = monotonic_milliseconds(io);
        event.bytes = bytes;
        event.length = action->length;
      }
      // GCOVR_EXCL_STOP
      result = rdt_sender_step(sender, &event, &next_actions);
      if (result != 0)
      {
        return 1;
      }
      current = next_actions;
      goto process_batch;
    }
    case RDT_ACTION_SEND_PACKET:
      // GCOVR_EXCL_START
      if (send_datagram(io, action->bytes, action->length) != 0)
      {
        memset(&event, 0, sizeof(event));
        event.type = RDT_SENDER_EVENT_IO_FAILURE;
        if (rdt_sender_step(sender, &event, &next_actions) != 0)
        {
          return 1;
        }
        current = next_actions;
        goto process_batch;
      }
      // GCOVR_EXCL_STOP
      break;
    case RDT_ACTION_WAIT_UNTIL:
      memset(&event, 0, sizeof(event));
      result = receive_protocol_packet(io, action->deadline_ms,
                                       &event.packet);
      event.now_ms = monotonic_milliseconds(io);
      event.type = result == 0 ? RDT_SENDER_EVENT_ACK
                               : RDT_SENDER_EVENT_TIMER;
      // GCOVR_EXCL_START
      if (result < 0)
      {
        event.type = RDT_SENDER_EVENT_IO_FAILURE;
      }
      // GCOVR_EXCL_STOP
      if (rdt_sender_step(sender, &event, &next_actions) != 0)
      {
        return 1;
      }
      current = next_actions;
      goto process_batch;
    case RDT_ACTION_COMPLETE:
      return 0;
    case RDT_ACTION_FAILED:
      if (action->status == 2)
      {
        fprintf(stderr, "%s\n", sender->finishing != 0
                                    ? "FIN was not acknowledged after 10 timeouts"
                                    : "DATA was not acknowledged after 10 timeouts");
      }
      return action->status;
    case RDT_ACTION_NONE:
    case RDT_ACTION_WRITE_FILE:
    case RDT_ACTION_FLUSH_FILE:
    case RDT_ACTION_CLOSE_FILE:
      return 1;
    default:
      return 1;
    }
  }
  return 0;
}

int rdt_io_send(struct rdt_io_context *context, uint64_t file_size,
                const struct rdt_options *options)
{
  struct rdt_sender sender;
  struct rdt_protocol_actions actions;

  if (context == NULL || options == NULL ||
      rdt_sender_init(&sender, file_size, options->window_size,
                      options->timeout_ms, monotonic_milliseconds(context),
                      &actions) != 0)
  {
    fprintf(stderr, "Invalid sender protocol configuration\n");
    return 1;
  }
  return sender_dispatch(context, &sender, &actions);
}

static int receiver_dispatch(struct rdt_io_context *io,
                             struct rdt_receiver *receiver,
                             const struct rdt_protocol_actions *actions)
{
  struct rdt_protocol_actions current = *actions;
  size_t i;

process_batch:
  for (i = 0U; i < current.count; ++i)
  {
    const struct rdt_protocol_action *action = &current.items[i];
    struct rdt_protocol_actions next_actions;
    struct rdt_receiver_event event;
    int result;

    switch (action->type)
    {
    case RDT_ACTION_WRITE_FILE:
      result = write_file(io, action->bytes, action->length);
      memset(&event, 0, sizeof(event));
      // GCOVR_EXCL_START
      event.type = result == 0 ? RDT_RECEIVER_EVENT_FILE_OK
                               : RDT_RECEIVER_EVENT_IO_FAILURE;
      // GCOVR_EXCL_STOP
      event.now_ms = monotonic_milliseconds(io);
      if (rdt_receiver_step(receiver, &event, &next_actions) != 0)
      {
        return 1;
      }
      current = next_actions;
      goto process_batch;
    case RDT_ACTION_FLUSH_FILE:
    case RDT_ACTION_CLOSE_FILE:
      result = action->type == RDT_ACTION_FLUSH_FILE
                   ? flush_file(io)
                   : close_file(io);
      memset(&event, 0, sizeof(event));
      // GCOVR_EXCL_START
      event.type = result == 0 ? RDT_RECEIVER_EVENT_FILE_OK
                               : RDT_RECEIVER_EVENT_IO_FAILURE;
      // GCOVR_EXCL_STOP
      event.now_ms = monotonic_milliseconds(io);
      if (rdt_receiver_step(receiver, &event, &next_actions) != 0)
      {
        return 1;
      }
      current = next_actions;
      goto process_batch;
    case RDT_ACTION_SEND_PACKET:
      // GCOVR_EXCL_START
      if (send_datagram(io, action->bytes, action->length) != 0)
      {
        memset(&event, 0, sizeof(event));
        event.type = RDT_RECEIVER_EVENT_IO_FAILURE;
        if (rdt_receiver_step(receiver, &event, &next_actions) != 0)
        {
          return 1;
        }
        current = next_actions;
        goto process_batch;
      }
      // GCOVR_EXCL_STOP
      break;
    case RDT_ACTION_WAIT_UNTIL:
      memset(&event, 0, sizeof(event));
      result = receive_protocol_packet(io, action->deadline_ms,
                                       &event.packet);
      event.now_ms = monotonic_milliseconds(io);
      event.type = result == 0 ? RDT_RECEIVER_EVENT_PACKET
                               : RDT_RECEIVER_EVENT_TIMER;
      // GCOVR_EXCL_START
      if (result < 0)
      {
        event.type = RDT_RECEIVER_EVENT_IO_FAILURE;
      }
      // GCOVR_EXCL_STOP
      if (rdt_receiver_step(receiver, &event, &next_actions) != 0)
      {
        return 1;
      }
      current = next_actions;
      goto process_batch;
    case RDT_ACTION_COMPLETE:
      return 0;
    case RDT_ACTION_FAILED:
      if (action->status == 2)
      {
        fprintf(stderr, "Receiver timed out without a valid packet\n");
      }
      return action->status;
    case RDT_ACTION_NONE:
    case RDT_ACTION_READ_FILE:
      return 1;
    default:
      return 1;
    }
  }
  return 0;
}
```

### rdt_protocol.c

```c

#include "rdt_protocol.h"

#include <string.h>

#define DATA_MAX_TIMEOUTS 10U
#define FIN_MAX_TIMEOUTS 10U
#define FIN_LINGER_MS UINT64_C(2000)
#define RECEIVE_IDLE_TIMEOUT_MS UINT64_C(30000)

static void clear_actions(struct rdt_protocol_actions *actions)
{
  actions->count = 0U;
}

static struct rdt_protocol_action *add_action(
    struct rdt_protocol_actions *actions,
    enum rdt_protocol_action_type type)
{
  struct rdt_protocol_action *action;

  action = &actions->items[actions->count++];
  memset(action, 0, sizeof(*action));
  action->type = type;
  return action;
}

static void add_wait(struct rdt_protocol_actions *actions, uint64_t deadline)
{
  struct rdt_protocol_action *action =
      add_action(actions, RDT_ACTION_WAIT_UNTIL);
  action->deadline_ms = deadline;
}

static void add_packet(struct rdt_protocol_actions *actions,
                       enum rdt_packet_type type, uint32_t sequence,
                       const uint8_t *payload, uint32_t payload_length)
{
  struct rdt_protocol_action *action =
      add_action(actions, RDT_ACTION_SEND_PACKET);

  action->length = rdt_encode_packet(action->bytes, sizeof(action->bytes),
                                     type, sequence, payload, payload_length);
}

static void add_status_action(struct rdt_protocol_actions *actions,
                              enum rdt_protocol_action_type type, int status)
{
  struct rdt_protocol_action *action = add_action(actions, type);
  action->status = status;
}

static void sender_request_read(struct rdt_sender *sender,
                                struct rdt_protocol_actions *actions)
{
  uint64_t offset;
  uint64_t remaining;
  size_t length;
  struct rdt_protocol_action *action;

  if (sender->read_pending != 0 || sender->next_sequence >= sender->total_packets ||
      sender->next_sequence - sender->base >= sender->window_size)
  {
    return;
  }
  offset = sender->next_sequence * (uint64_t) RDT_MAX_PAYLOAD;
  remaining = sender->file_size - offset;
  length = remaining > (uint64_t) RDT_MAX_PAYLOAD
               ? (size_t) RDT_MAX_PAYLOAD
               : (size_t) remaining;
  action = add_action(actions, RDT_ACTION_READ_FILE);
  action->length = length;
  sender->read_pending = 1;
}

static void sender_wait_or_read(struct rdt_sender *sender, uint64_t now_ms,
                                struct rdt_protocol_actions *actions)
{
  if (sender->base == sender->total_packets)
  {
    sender->finishing = 1;
    sender->timer_active = 1;
    sender->deadline_ms = now_ms + (uint64_t) sender->timeout_ms;
    add_packet(actions, RDT_FIN, (uint32_t) sender->total_packets, NULL, 0U);
    add_wait(actions, sender->deadline_ms);
    return;
  }
  sender_request_read(sender, actions);
  if (sender->timer_active != 0)
  {
    add_wait(actions, sender->deadline_ms);
  }
}

int rdt_sender_init(struct rdt_sender *sender, uint64_t file_size,
                    unsigned int window_size, unsigned int timeout_ms,
                    uint64_t now_ms, struct rdt_protocol_actions *actions)
{
  if (sender == NULL || actions == NULL || window_size == 0U ||
      window_size > 64U || timeout_ms == 0U || timeout_ms > 60000U)
  {
    return -1;
  }
  memset(sender, 0, sizeof(*sender));
  clear_actions(actions);
  sender->file_size = file_size;
  sender->total_packets =
      file_size / (uint64_t) RDT_MAX_PAYLOAD +
      (file_size % (uint64_t) RDT_MAX_PAYLOAD == 0U ? 0U : 1U);
  if (sender->total_packets >= (uint64_t) UINT32_MAX)
  {
    return -1;
  }
  sender->window_size = window_size;
  sender->timeout_ms = timeout_ms;
  if (sender->total_packets == 0U)
  {
    sender_wait_or_read(sender, now_ms, actions);
    return 0;
  }
  sender_request_read(sender, actions);
  return 0;
}

static void sender_emit_outstanding(struct rdt_sender *sender,
                                    struct rdt_protocol_actions *actions)
{
  uint64_t sequence;

  for (sequence = sender->base; sequence < sender->next_sequence; ++sequence)
  {
    struct rdt_sender_slot *slot =
        &sender->slots[sequence % (uint64_t) sender->window_size];
    struct rdt_protocol_action *action =
        add_action(actions, RDT_ACTION_SEND_PACKET);
    memcpy(action->bytes, slot->bytes, slot->length);
    action->length = slot->length;
  }
}

int rdt_sender_step(struct rdt_sender *sender,
                    const struct rdt_sender_event *event,
                    struct rdt_protocol_actions *actions)
{
  if (sender == NULL || event == NULL || actions == NULL)
  {
    return -1;
  }
  clear_actions(actions);
  if (sender->done != 0)
  {
    add_status_action(actions, RDT_ACTION_COMPLETE, 0);
    return 0;
  }
  if (event->type == RDT_SENDER_EVENT_IO_FAILURE)
  {
    sender->done = 1;
    add_status_action(actions, RDT_ACTION_FAILED, 1);
    return 0;
  }

  if (sender->finishing != 0)
  {
    if (event->type == RDT_SENDER_EVENT_ACK &&
        event->packet.type == RDT_ACK &&
        event->packet.payload_length == 0U &&
        (uint64_t) event->packet.sequence == sender->total_packets + 1U)
    {
      sender->done = 1;
      add_status_action(actions, RDT_ACTION_COMPLETE, 0);
      return 0;
    }
    if (event->type == RDT_SENDER_EVENT_TIMER &&
        event->now_ms >= sender->deadline_ms)
    {
      ++sender->consecutive_timeouts;
      if (sender->consecutive_timeouts >= FIN_MAX_TIMEOUTS)
      {
        sender->done = 1;
        add_status_action(actions, RDT_ACTION_FAILED, 2);
        return 0;
      }
      sender->deadline_ms = event->now_ms + (uint64_t) sender->timeout_ms;
      add_packet(actions, RDT_FIN, (uint32_t) sender->total_packets, NULL, 0U);
    }
    add_wait(actions, sender->deadline_ms);
    return 0;
  }

  if (event->type == RDT_SENDER_EVENT_DATA_READY)
  {
    uint64_t offset;
    uint64_t remaining;
    size_t expected_length;
    struct rdt_sender_slot *slot;

    if (sender->read_pending == 0 || event->bytes == NULL)
    {
      return -1;
    }
    offset = sender->next_sequence * (uint64_t) RDT_MAX_PAYLOAD;
    remaining = sender->file_size - offset;
    expected_length = remaining > (uint64_t) RDT_MAX_PAYLOAD
                          ? (size_t) RDT_MAX_PAYLOAD
                          : (size_t) remaining;
    if (event->length != expected_length)
    {
      sender->done = 1;
      add_status_action(actions, RDT_ACTION_FAILED, 1);
      return 0;
    }
    slot = &sender->slots[sender->next_sequence %
                          (uint64_t) sender->window_size];
    slot->length = rdt_encode_packet(
        slot->bytes, sizeof(slot->bytes), RDT_DATA,
        (uint32_t) sender->next_sequence, event->bytes,
        (uint32_t) event->length);
    sender->read_pending = 0;
    add_packet(actions, RDT_DATA, (uint32_t) sender->next_sequence,
               event->bytes, (uint32_t) event->length);
    ++sender->next_sequence;
    if (sender->timer_active == 0)
    {
      sender->timer_active = 1;
      sender->deadline_ms = event->now_ms + (uint64_t) sender->timeout_ms;
    }
    sender_wait_or_read(sender, event->now_ms, actions);
    return 0;
  }

  if (event->type == RDT_SENDER_EVENT_ACK &&
      event->packet.type == RDT_ACK &&
      event->packet.payload_length == 0U &&
      (uint64_t) event->packet.sequence > sender->base &&
      (uint64_t) event->packet.sequence <= sender->next_sequence)
  {
    sender->base = event->packet.sequence;
    sender->consecutive_timeouts = 0U;
    if (sender->base < sender->next_sequence)
    {
      sender->timer_active = 1;
      sender->deadline_ms = event->now_ms + (uint64_t) sender->timeout_ms;
    }
    else
    {
      sender->timer_active = 0;
    }
    sender_wait_or_read(sender, event->now_ms, actions);
    return 0;
  }

  if (event->type == RDT_SENDER_EVENT_TIMER &&
      sender->timer_active != 0 && event->now_ms >= sender->deadline_ms)
  {
    ++sender->consecutive_timeouts;
    if (sender->consecutive_timeouts >= DATA_MAX_TIMEOUTS)
    {
      sender->done = 1;
      add_status_action(actions, RDT_ACTION_FAILED, 2);
      return 0;
    }
    sender_emit_outstanding(sender, actions);
    sender->deadline_ms = event->now_ms + (uint64_t) sender->timeout_ms;
  }
  sender_wait_or_read(sender, event->now_ms, actions);
  return 0;
}

static void receiver_ack(struct rdt_receiver *receiver,
                         struct rdt_protocol_actions *actions)
{
  add_packet(actions, RDT_ACK, (uint32_t) receiver->expected, NULL, 0U);
}

int rdt_receiver_init(struct rdt_receiver *receiver, uint64_t now_ms,
                      struct rdt_protocol_actions *actions)
{
  if (receiver == NULL || actions == NULL)
  {
    return -1;
  }
  memset(receiver, 0, sizeof(*receiver));
  clear_actions(actions);
  receiver->phase = RDT_RECEIVER_ACTIVE;
  receiver->idle_deadline_ms = now_ms + RECEIVE_IDLE_TIMEOUT_MS;
  add_wait(actions, receiver->idle_deadline_ms);
  return 0;
}

int rdt_receiver_step(struct rdt_receiver *receiver,
                      const struct rdt_receiver_event *event,
                      struct rdt_protocol_actions *actions)
{
  if (receiver == NULL || event == NULL || actions == NULL)
  {
    return -1;
  }
  clear_actions(actions);
  if (event->type == RDT_RECEIVER_EVENT_IO_FAILURE)
  {
    receiver->phase = RDT_RECEIVER_FAILED;
    add_status_action(actions, RDT_ACTION_FAILED, 1);
    return 0;
  }
  if (receiver->phase == RDT_RECEIVER_DONE)
  {
    add_status_action(actions, RDT_ACTION_COMPLETE, 0);
    return 0;
  }
  if (receiver->phase == RDT_RECEIVER_FAILED)
  {
    add_status_action(actions, RDT_ACTION_FAILED, 1);
    return 0;
  }
  if (receiver->phase == RDT_RECEIVER_ACTIVE &&
      event->type == RDT_RECEIVER_EVENT_TIMER &&
      event->now_ms >= receiver->idle_deadline_ms)
  {
    receiver->phase = RDT_RECEIVER_FAILED;
    add_status_action(actions, RDT_ACTION_FAILED, 2);
    return 0;
  }
  if (receiver->phase == RDT_RECEIVER_LINGER &&
      event->type == RDT_RECEIVER_EVENT_TIMER &&
      event->now_ms >= receiver->linger_deadline_ms)
  {
    receiver->phase = RDT_RECEIVER_DONE;
    add_status_action(actions, RDT_ACTION_COMPLETE, 0);
    return 0;
  }

  if (event->type == RDT_RECEIVER_EVENT_PACKET &&
      receiver->phase == RDT_RECEIVER_ACTIVE)
  {
    receiver->idle_deadline_ms = event->now_ms + RECEIVE_IDLE_TIMEOUT_MS;
    if (event->packet.type == RDT_DATA)
    {
      if ((uint64_t) event->packet.sequence != receiver->expected ||
          event->packet.payload_length == 0U ||
          event->packet.payload_length > RDT_MAX_PAYLOAD)
      {
        receiver_ack(receiver, actions);
      }
      else
      {
        memcpy(receiver->pending_payload, event->packet.payload,
               event->packet.payload_length);
        receiver->pending_length = event->packet.payload_length;
        receiver->phase = RDT_RECEIVER_WRITING;
        struct rdt_protocol_action *action =
            add_action(actions, RDT_ACTION_WRITE_FILE);
        memcpy(action->bytes, receiver->pending_payload,
               receiver->pending_length);
        action->length = receiver->pending_length;
      }
    }
    else if (event->packet.type == RDT_FIN)
    {
      if ((uint64_t) event->packet.sequence != receiver->expected)
      {
        receiver_ack(receiver, actions);
      }
      else
      {
        receiver->fin_sequence = event->packet.sequence;
        receiver->phase = RDT_RECEIVER_FLUSHING;
        (void) add_action(actions, RDT_ACTION_FLUSH_FILE);
      }
    }
    if (receiver->phase == RDT_RECEIVER_ACTIVE)
    {
      add_wait(actions, receiver->idle_deadline_ms);
    }
    return 0;
  }

  if (receiver->phase == RDT_RECEIVER_LINGER &&
      event->type == RDT_RECEIVER_EVENT_PACKET)
  {
    if (event->packet.type == RDT_FIN &&
        event->packet.sequence == receiver->fin_sequence)
    {
      receiver->expected = (uint64_t) receiver->fin_sequence + 1U;
      receiver_ack(receiver, actions);
    }
    add_wait(actions, receiver->linger_deadline_ms);
    return 0;
  }

  if (event->type == RDT_RECEIVER_EVENT_FILE_OK)
  {
    if (receiver->phase == RDT_RECEIVER_WRITING)
    {
      ++receiver->expected;
      receiver->phase = RDT_RECEIVER_ACTIVE;
      receiver->idle_deadline_ms = event->now_ms + RECEIVE_IDLE_TIMEOUT_MS;
      receiver_ack(receiver, actions);
      add_wait(actions, receiver->idle_deadline_ms);
      return 0;
    }
    if (receiver->phase == RDT_RECEIVER_FLUSHING)
    {
      receiver->phase = RDT_RECEIVER_CLOSING;
      (void) add_action(actions, RDT_ACTION_CLOSE_FILE);
      return 0;
    }
    if (receiver->phase == RDT_RECEIVER_CLOSING)
    {
      receiver->expected = (uint64_t) receiver->fin_sequence + 1U;
      receiver->phase = RDT_RECEIVER_LINGER;
      receiver->linger_deadline_ms = event->now_ms + FIN_LINGER_MS;
      receiver_ack(receiver, actions);
      add_wait(actions, receiver->linger_deadline_ms);
      return 0;
    }
  }

  if (event->type == RDT_RECEIVER_EVENT_TIMER)
  {
    add_wait(actions, receiver->phase == RDT_RECEIVER_LINGER
                          ? receiver->linger_deadline_ms
                          : receiver->idle_deadline_ms);
  }
  return 0;
}
```

```c
static int make_relay_socket(unsigned short *port)
{
  struct sockaddr_in address;
  socklen_t address_length = (socklen_t) sizeof(address);
  int socket_fd = socket(AF_INET, SOCK_DGRAM, 0);

  if (socket_fd < 0)
  {
    return -1;
  }
  memset(&address, 0, sizeof(address));
  address.sin_family = AF_INET;
  address.sin_addr.s_addr = htonl(INADDR_LOOPBACK);
  address.sin_port = htons(0U);
  if (bind(socket_fd, (struct sockaddr *) &address, sizeof(address)) != 0 ||
      getsockname(socket_fd, (struct sockaddr *) &address, &address_length) !=
          0)
  {
    (void) close(socket_fd);
    return -1;
  }
  *port = ntohs(address.sin_port);
  return socket_fd;
}

static int receive_relay_datagram(int socket_fd, uint8_t *packet,
                                 size_t capacity,
                                 struct sockaddr_in *source, int timeout_ms)
{
  struct pollfd descriptor = {
    .fd = socket_fd,
    .events = POLLIN,
    .revents = 0
  };
  socklen_t source_length = (socklen_t) sizeof(*source);
  int poll_result = poll(&descriptor, 1U, timeout_ms);
  ssize_t received;

  if (poll_result <= 0 || (descriptor.revents & POLLIN) == 0)
  {
    return -1;
  }
  received = recvfrom(socket_fd, packet, capacity, 0,
                      (struct sockaddr *) source, &source_length);
  if (received < 0)
  {
    return -1;
  }
  return (int) received;
}

static int send_relay_datagram(int socket_fd, const uint8_t *packet,
                               size_t length,
                               const struct sockaddr_in *destination)
{
  ssize_t sent = sendto(socket_fd, packet, length, 0,
                        (const struct sockaddr *) destination,
                        (socklen_t) sizeof(*destination));
  return sent >= 0 && (size_t) sent == length ? 0 : -1;
}

static int same_address(const struct sockaddr_in *left,
                        const struct sockaddr_in *right)
{
  return left->sin_addr.s_addr == right->sin_addr.s_addr &&
         left->sin_port == right->sin_port;
}

static void handle_test_relay_packet(int socket_fd,
                                     struct relay_test_state *state,
                                     const uint8_t *packet, size_t length,
                                     const struct sockaddr_in *source)
{
  uint8_t modified[RDT_MAX_PACKET];
  char hello[128];
  struct sockaddr_in *destination;

  if (length >= 6U && memcmp(packet, "HELLO ", 6U) == 0)
  {
    if (length >= sizeof(hello))
    {
      return;
    }
    memcpy(hello, packet, length);
    hello[length] = '\0';
    if (strstr(hello, " recv") != NULL)
    {
      state->receiver = *source;
      state->has_receiver = 1;
    }
    else if (strstr(hello, " send ") != NULL)
    {
      state->sender = *source;
      state->has_sender = 1;
    }
    (void) send_relay_datagram(socket_fd, (const uint8_t *) "OK", 2U, source);
    return;
  }

  if (state->has_sender != 0 && same_address(source, &state->sender))
  {
    enum rdt_packet_type type;
    uint32_t sequence;
    const uint8_t *payload;
    uint32_t payload_length;

    if (rdt_decode_packet(packet, length, &type, &sequence, &payload,
                          &payload_length) != 0)
    {
      return;
    }
    destination = &state->receiver;
    if (type == RDT_DATA && sequence == 0U && state->drop_first_data != 0)
    {
      state->drop_first_data = 0;
      return;
    }
    if (type == RDT_DATA && sequence == 1U &&
        state->corrupt_first_data != 0)
    {
      state->corrupt_first_data = 0;
      memcpy(modified, packet, length);
      modified[RDT_HEADER_SIZE] ^= 1U;
      (void) send_relay_datagram(socket_fd, modified, length, destination);
      return;
    }
    if (type == RDT_DATA && sequence == 2U && state->duplicate_data != 0)
    {
      state->duplicate_data = 0;
      (void) send_relay_datagram(socket_fd, packet, length, destination);
      (void) send_relay_datagram(socket_fd, packet, length, destination);
      return;
    }
    if (type == RDT_FIN)
    {
      state->fin_sequence = sequence;
    }
  }
  else if (state->has_receiver != 0 &&
           same_address(source, &state->receiver))
  {
    enum rdt_packet_type type;
    uint32_t sequence;

    if (rdt_decode_packet(packet, length, &type, &sequence, NULL, NULL) != 0)
    {
      return;
    }
    destination = &state->sender;
    if (type == RDT_ACK && sequence == state->fin_sequence + 1U &&
        (state->drop_fin_ack != 0 || state->drop_all_fin_acks != 0))
    {
      if (state->drop_all_fin_acks == 0)
      {
        state->drop_fin_ack = 0;
      }
      return;
    }
  }
  else
  {
    return;
  }

  if (state->has_receiver != 0 && state->has_sender != 0)
  {
    (void) send_relay_datagram(socket_fd, packet, length, destination);
  }
}

static int write_test_file(const char *path, const uint8_t *bytes,
                           size_t length)
{
  FILE *file = fopen(path, "wb");
  int result = 0;

  if (file == NULL)
  {
    return -1;
  }
  if (length > 0U && fwrite(bytes, 1U, length, file) != length)
  {
    result = -1;
  }
  if (fclose(file) != 0)
  {
    result = -1;
  }
  return result;
}

static void finish_child(pid_t *child, int *status)
{
  if (*child > 0)
  {
    (void) waitpid(*child, status, 0);
    *child = -1;
  }
}

static int run_transfer_case(const uint8_t *contents, size_t content_length,
                            const char *session, int inject_faults,
                            int drop_all_fin_acks)
{
  struct relay_test_state relay_state = {0};
  struct rdt_options options = {
    .session = session,
    .window_size = 4U,
    .timeout_ms = 50U,
    .port = 0U,
    .loss = 0.0,
    .corrupt = 0.0,
    .duplicate = 0.0
  };
  uint8_t datagram[RDT_MAX_PACKET];
  unsigned short relay_port = 0U;
  char input_path[] = "/tmp/cs425-rdt-input-XXXXXX";
  char output_path[] = "/tmp/cs425-rdt-output-XXXXXX";
  int input_fd = -1;
  int output_fd = -1;
  int relay_fd = -1;
  int sender_status = 0;
  int receiver_status = 0;
  int datagram_length;
  pid_t receiver_pid = -1;
  pid_t sender_pid = -1;
  uint64_t deadline;
  FILE *received_file = NULL;
  uint8_t *received_contents = NULL;
  size_t received_length = 0U;
  int result = -1;

  input_fd = mkstemp(input_path);
  output_fd = mkstemp(output_path);
  if (input_fd < 0 || output_fd < 0)
  {
    goto cleanup;
  }
  (void) close(output_fd);
  output_fd = -1;
  if (write_test_file(input_path, contents, content_length) != 0 ||
      unlink(output_path) != 0)
  {
    goto cleanup;
  }
  relay_fd = make_relay_socket(&relay_port);
  if (relay_fd < 0)
  {
    goto cleanup;
  }
  options.port = relay_port;
  relay_state.drop_first_data = inject_faults;
  relay_state.corrupt_first_data = inject_faults;
  relay_state.duplicate_data = inject_faults;
  relay_state.drop_fin_ack = inject_faults;
  relay_state.drop_all_fin_acks = drop_all_fin_acks;

  (void) fflush(NULL);
  receiver_pid = fork();
  if (receiver_pid < 0)
  {
    goto cleanup;
  }
  if (receiver_pid == 0)
  {
    exit(rdt_receive_file("127.0.0.1", relay_port, output_path, &options));
  }

  datagram_length = receive_relay_datagram(
      relay_fd, datagram, sizeof(datagram), &relay_state.receiver, 1000);
  if (datagram_length < 0 || datagram_length < 6 ||
      memcmp(datagram, "HELLO ", 6U) != 0)
  {
    goto cleanup;
  }
  relay_state.has_receiver = 1;
  if (send_relay_datagram(relay_fd, (const uint8_t *) "OK", 2U,
                          &relay_state.receiver) != 0)
  {
    goto cleanup;
  }

  sender_pid = fork();
  if (sender_pid < 0)
  {
    goto cleanup;
  }
  if (sender_pid == 0)
  {
    exit(rdt_send_file("127.0.0.1", relay_port, input_path, &options));
  }

  deadline = test_monotonic_milliseconds() + UINT64_C(8000);
  while (test_monotonic_milliseconds() < deadline &&
         (sender_pid > 0 || receiver_pid > 0))
  {
    struct sockaddr_in source;
    int remaining = 50;

    datagram_length = receive_relay_datagram(
        relay_fd, datagram, sizeof(datagram), &source, remaining);
    if (datagram_length >= 0)
    {
      handle_test_relay_packet(relay_fd, &relay_state, datagram,
                               (size_t) datagram_length, &source);
    }

    if (sender_pid > 0)
    {
      pid_t waited = waitpid(sender_pid, &sender_status, WNOHANG);
      if (waited == sender_pid)
      {
        sender_pid = -1;
      }
    }
    if (receiver_pid > 0)
    {
      pid_t waited = waitpid(receiver_pid, &receiver_status, WNOHANG);
      if (waited == receiver_pid)
      {
        receiver_pid = -1;
      }
    }
  }

  if (sender_pid > 0 || receiver_pid > 0)
  {
    goto cleanup;
  }
  if (!WIFEXITED(sender_status) ||
      WEXITSTATUS(sender_status) != (drop_all_fin_acks != 0 ? 2 : 0) ||
      !WIFEXITED(receiver_status) || WEXITSTATUS(receiver_status) != 0)
  {
    goto cleanup;
  }
  if (inject_faults != 0 &&
      (relay_state.drop_first_data != 0 ||
       relay_state.corrupt_first_data != 0 ||
       relay_state.duplicate_data != 0 || relay_state.drop_fin_ack != 0))
  {
    goto cleanup;
  }
  if (drop_all_fin_acks != 0 && relay_state.drop_all_fin_acks == 0)
  {
    goto cleanup;
  }

  received_file = fopen(output_path, "rb");
  if (received_file == NULL)
  {
    goto cleanup;
  }
  if (content_length > 0U)
  {
    received_contents = malloc(content_length);
    if (received_contents == NULL)
    {
      goto cleanup;
    }
    received_length = fread(received_contents, 1U, content_length,
                            received_file);
  }
  if (received_length != content_length ||
      (content_length > 0U &&
       memcmp(received_contents, contents, content_length) != 0))
  {
    goto cleanup;
  }
  result = 0;

cleanup:
  if (sender_pid > 0)
  {
    (void) kill(sender_pid, SIGTERM);
    finish_child(&sender_pid, &sender_status);
  }
  if (receiver_pid > 0)
  {
    (void) kill(receiver_pid, SIGTERM);
    finish_child(&receiver_pid, &receiver_status);
  }
  if (received_file != NULL)
  {
    (void) fclose(received_file);
  }
  free(received_contents);
  if (relay_fd >= 0)
  {
    (void) close(relay_fd);
  }
  if (input_fd >= 0)
  {
    (void) close(input_fd);
  }
  (void) unlink(input_path);
  (void) unlink(output_path);
  return result;
}

static void test_sender_window_cumulative_ack_and_timeout(void)
{
  struct rdt_sender sender;
  struct rdt_protocol_actions actions;
  struct rdt_sender_event event = {0};
  uint8_t payload[RDT_MAX_PAYLOAD] = {0};
  unsigned int sequence;
  enum rdt_packet_type type;
  uint32_t packet_sequence;

  TEST_ASSERT_EQUAL_INT(0, rdt_sender_init(&sender, sizeof(payload) * 5U, 4U,
                                           10U, 0U, &actions));
  event.type = RDT_SENDER_EVENT_DATA_READY;
  event.bytes = payload;
  event.length = sizeof(payload);
  for (sequence = 0U; sequence < 4U; ++sequence)
  {
    event.now_ms = (uint64_t) sequence;
    TEST_ASSERT_EQUAL_INT(0, rdt_sender_step(&sender, &event, &actions));
    TEST_ASSERT_EQUAL_INT(RDT_ACTION_SEND_PACKET, actions.items[0].type);
  }
  TEST_ASSERT_EQUAL_UINT64(4U, sender.next_sequence);
  TEST_ASSERT_EQUAL_UINT64(0U, sender.base);

  event.type = RDT_SENDER_EVENT_ACK;
  event.packet.type = RDT_ACK;
  event.packet.sequence = 0U;
  event.now_ms = 4U;
  TEST_ASSERT_EQUAL_INT(0, rdt_sender_step(&sender, &event, &actions));
  TEST_ASSERT_EQUAL_UINT64(0U, sender.base);
  TEST_ASSERT_EQUAL_INT(RDT_ACTION_WAIT_UNTIL, actions.items[0].type);

  event.packet.sequence = 3U;
  event.now_ms = 5U;
  TEST_ASSERT_EQUAL_INT(0, rdt_sender_step(&sender, &event, &actions));
  TEST_ASSERT_EQUAL_UINT64(3U, sender.base);
  TEST_ASSERT_EQUAL_INT(RDT_ACTION_READ_FILE, actions.items[0].type);

  event.type = RDT_SENDER_EVENT_DATA_READY;
  event.now_ms = 6U;
  event.bytes = payload;
  event.length = sizeof(payload);
  TEST_ASSERT_EQUAL_INT(0, rdt_sender_step(&sender, &event, &actions));
  TEST_ASSERT_EQUAL_INT(RDT_ACTION_SEND_PACKET, actions.items[0].type);
  event.type = RDT_SENDER_EVENT_TIMER;
  event.now_ms = 16U;
  TEST_ASSERT_EQUAL_INT(0, rdt_sender_step(&sender, &event, &actions));
  TEST_ASSERT_EQUAL_UINT(3U, actions.count);
  for (sequence = 0U; sequence < 2U; ++sequence)
  {
    TEST_ASSERT_EQUAL_INT(RDT_ACTION_SEND_PACKET, actions.items[sequence].type);
    TEST_ASSERT_EQUAL_INT(
        0, rdt_decode_packet(actions.items[sequence].bytes,
                             actions.items[sequence].length, &type,
                             &packet_sequence, NULL, NULL));
    TEST_ASSERT_EQUAL_UINT32(sequence + 3U, packet_sequence);
  }
  TEST_ASSERT_EQUAL_INT(RDT_ACTION_WAIT_UNTIL, actions.items[2].type);
}

  TEST_ASSERT_NOT_EQUAL(

      0, rdt_receiver_step(&receiver, NULL, &actions));
  TEST_ASSERT_NOT_EQUAL(
      0, rdt_receiver_step(&receiver, &receiver_event, NULL));

  sender_event.type = RDT_SENDER_EVENT_DATA_READY;
  sender_event.bytes = NULL;
  sender_event.length = 1U;

  TEST_ASSERT_EQUAL_INT(0, rdt_sender_step(&sender, &sender_event, &actions));
  TEST_ASSERT_NOT_EQUAL(0, rdt_sender_step(&sender, &sender_event, &actions));
  sender_event.type = RDT_SENDER_EVENT_IO_FAILURE;
  TEST_ASSERT_EQUAL_INT(0, rdt_sender_step(&sender, &sender_event, &actions));
  TEST_ASSERT_EQUAL_INT(RDT_ACTION_FAILED, actions.items[0].type);
}

#define MEMORY_CHANNEL_CAPACITY 128U

struct memory_channel
{
  uint8_t packets[MEMORY_CHANNEL_CAPACITY][RDT_MAX_PACKET];
  size_t lengths[MEMORY_CHANNEL_CAPACITY];
  size_t head;
  size_t count;
  uint32_t random_state;
  unsigned int dropped;
  unsigned int corrupted;
  unsigned int duplicated;
};

struct state_machine_transfer
{
  struct rdt_sender sender;
  struct rdt_receiver receiver;
  struct memory_channel sender_to_receiver;
  struct memory_channel receiver_to_sender;
  const uint8_t *source;
  size_t source_length;
  size_t source_offset;
  uint8_t destination[16384];
  size_t destination_length;
  uint64_t now_ms;
  int sender_done;
  int receiver_done;
  int status;
};

static uint32_t memory_channel_random(struct memory_channel *channel)
{
  uint32_t value = channel->random_state;
  value ^= value << 13;
  value ^= value >> 17;
  value ^= value << 5;
  channel->random_state = value;
  return value;
}

static int memory_channel_enqueue(struct memory_channel *channel,
                                  const uint8_t *packet, size_t length)
{
  uint8_t impaired[RDT_MAX_PACKET];
  unsigned int copies;
  unsigned int copy;

  if (length > sizeof(impaired) ||
      memory_channel_random(channel) % 100U < 20U)
  {
    ++channel->dropped;
    return 0;
  }
  memcpy(impaired, packet, length);
  if (memory_channel_random(channel) % 100U < 20U)
  {
    impaired[length - 1U] ^= 1U;
    ++channel->corrupted;
  }
  copies = memory_channel_random(channel) % 100U < 20U ? 2U : 1U;
  if (copies == 2U)
  {
    ++channel->duplicated;
  }
  if (copies > MEMORY_CHANNEL_CAPACITY - channel->count)
  {
    return -1;
  }
  for (copy = 0U; copy < copies; ++copy)
  {
    size_t index = (channel->head + channel->count) %
                   MEMORY_CHANNEL_CAPACITY;
    memcpy(channel->packets[index], impaired, length);
    channel->lengths[index] = length;
    ++channel->count;
  }
  return 0;
}

static int memory_channel_pop(struct memory_channel *channel, uint8_t *packet,
                              size_t *length)
{
  if (channel->count == 0U)
  {
    return 0;
  }
  *length = channel->lengths[channel->head];
  memcpy(packet, channel->packets[channel->head], *length);
  channel->head = (channel->head + 1U) % MEMORY_CHANNEL_CAPACITY;
  --channel->count;
  return 1;
}

static int drive_sender_actions(
    struct state_machine_transfer *transfer,
    const struct rdt_protocol_actions *actions)
{
  size_t i;

  for (i = 0U; i < actions->count; ++i)
  {
    const struct rdt_protocol_action *action = &actions->items[i];
    struct rdt_protocol_actions next_actions;
    struct rdt_sender_event event = {0};

    if (action->type == RDT_ACTION_READ_FILE)
    {
      if (transfer->source_offset > transfer->source_length ||
          action->length > transfer->source_length -
                               transfer->source_offset)
      {
        return -1;
      }
      event.type = RDT_SENDER_EVENT_DATA_READY;
      event.now_ms = transfer->now_ms;
      event.bytes = transfer->source + transfer->source_offset;
      event.length = action->length;
      transfer->source_offset += action->length;
      if (rdt_sender_step(&transfer->sender, &event, &next_actions) != 0 ||
          drive_sender_actions(transfer, &next_actions) != 0)
      {
        return -1;
      }
      return 0;
    }
    if (action->type == RDT_ACTION_SEND_PACKET)
    {
      if (memory_channel_enqueue(&transfer->sender_to_receiver,
                                action->bytes, action->length) != 0)
      {
        return -1;
      }
    }
    else if (action->type == RDT_ACTION_COMPLETE)
    {
      transfer->sender_done = 1;
      transfer->status = action->status;
    }
    else if (action->type == RDT_ACTION_FAILED)
    {
      transfer->sender_done = 1;
      transfer->status = action->status;
    }
  }
  return 0;
}

static int drive_receiver_actions(
    struct state_machine_transfer *transfer,
    const struct rdt_protocol_actions *actions)
{
  size_t i;

  for (i = 0U; i < actions->count; ++i)
  {
    const struct rdt_protocol_action *action = &actions->items[i];
    struct rdt_protocol_actions next_actions;
    struct rdt_receiver_event event = {0};

    if (action->type == RDT_ACTION_WRITE_FILE)
    {
      if (action->length > sizeof(transfer->destination) -
                               transfer->destination_length)
      {
        return -1;
      }
      memcpy(transfer->destination + transfer->destination_length,
             action->bytes, action->length);
      transfer->destination_length += action->length;
      event.type = RDT_RECEIVER_EVENT_FILE_OK;
      event.now_ms = transfer->now_ms;
      if (rdt_receiver_step(&transfer->receiver, &event, &next_actions) != 0 ||
          drive_receiver_actions(transfer, &next_actions) != 0)
      {
        return -1;
      }
      return 0;
    }
    if (action->type == RDT_ACTION_FLUSH_FILE ||
        action->type == RDT_ACTION_CLOSE_FILE)
    {
      event.type = RDT_RECEIVER_EVENT_FILE_OK;
      event.now_ms = transfer->now_ms;
      if (rdt_receiver_step(&transfer->receiver, &event, &next_actions) != 0 ||
          drive_receiver_actions(transfer, &next_actions) != 0)
      {
        return -1;
      }
      return 0;
    }
    if (action->type == RDT_ACTION_SEND_PACKET)
    {
      if (memory_channel_enqueue(&transfer->receiver_to_sender,
                                action->bytes, action->length) != 0)
      {
        return -1;
      }
    }
    else if (action->type == RDT_ACTION_COMPLETE)
    {
      transfer->receiver_done = 1;
    }
    else if (action->type == RDT_ACTION_FAILED)
    {
      transfer->receiver_done = 1;
      transfer->status = action->status;
    }
  }
  return 0;
}

static int drive_sender_packet(struct state_machine_transfer *transfer,
                               const uint8_t *bytes, size_t length)
{
  struct rdt_sender_event event = {0};
  struct rdt_protocol_actions actions;
  const uint8_t *payload = NULL;

  if (rdt_decode_packet(bytes, length, &event.packet.type,
                        &event.packet.sequence, &payload,
                        &event.packet.payload_length) != 0)
  {
    return 0;
  }
  if (event.packet.type != RDT_ACK)
  {
    return 0;
  }
  event.type = RDT_SENDER_EVENT_ACK;
  event.now_ms = transfer->now_ms;
  (void) payload;
  if (rdt_sender_step(&transfer->sender, &event, &actions) != 0)
  {
    return -1;
  }
  return drive_sender_actions(transfer, &actions);
}

static int drive_receiver_packet(struct state_machine_transfer *transfer,
                                 const uint8_t *bytes, size_t length)
{
  struct rdt_receiver_event event = {0};
  struct rdt_protocol_actions actions;
  const uint8_t *payload = NULL;

  if (rdt_decode_packet(bytes, length, &event.packet.type,
                        &event.packet.sequence, &payload,
                        &event.packet.payload_length) != 0)
  {
    return 0;
  }
  if (event.packet.payload_length > 0U)
  {
    memcpy(event.packet.payload, payload, event.packet.payload_length);
  }
  event.type = RDT_RECEIVER_EVENT_PACKET;
  event.now_ms = transfer->now_ms;
  if (rdt_receiver_step(&transfer->receiver, &event, &actions) != 0)
  {
    return -1;
  }
  return drive_receiver_actions(transfer, &actions);
}

static int run_seeded_state_machine_transfer(uint32_t seed,
                                            unsigned int impairment_counts[6])
{
  struct state_machine_transfer transfer = {0};
  struct rdt_protocol_actions actions;
  uint8_t packet[RDT_MAX_PACKET];
  size_t packet_length;
  size_t iteration;
  size_t i;
  struct rdt_sender_event sender_event = {0};
  struct rdt_receiver_event receiver_event = {0};
  const size_t source_length = 7U * RDT_MAX_PAYLOAD + 321U;
  uint8_t source[7U * RDT_MAX_PAYLOAD + 321U];

  transfer.source = source;
  transfer.source_length = source_length;
  transfer.sender_to_receiver.random_state = seed;
  transfer.receiver_to_sender.random_state = seed ^ UINT32_C(0x9e3779b9);
  for (i = 0U; i < source_length; ++i)
  {
    source[i] = (uint8_t) ((i * 73U + (size_t) seed) & 0xffU);
  }
  if (rdt_sender_init(&transfer.sender, source_length, 4U, 25U, 0U,
                      &actions) != 0 ||
      drive_sender_actions(&transfer, &actions) != 0 ||
      rdt_receiver_init(&transfer.receiver, 0U, &actions) != 0)
  {
    return -1;
  }

  for (iteration = 0U; iteration < 100000U; ++iteration)
  {
    int progressed = 0;

    if (transfer.sender_to_receiver.count > 0U &&
        memory_channel_pop(&transfer.sender_to_receiver, packet,
                           &packet_length) != 0)
    {
      if (drive_receiver_packet(&transfer, packet, packet_length) != 0)
      {
        fprintf(stderr, "seed=%u receiver packet action failed at %zu\n",
                seed, iteration);
        return -1;
      }
      progressed = 1;
    }
    if (transfer.receiver_to_sender.count > 0U &&
        memory_channel_pop(&transfer.receiver_to_sender, packet,
                           &packet_length) != 0)
    {
      if (drive_sender_packet(&transfer, packet, packet_length) != 0)
      {
        fprintf(stderr, "seed=%u sender packet action failed at %zu\n",
                seed, iteration);
        return -1;
      }
      progressed = 1;
    }
    if (transfer.sender_done != 0 && transfer.receiver_done != 0)
    {
      break;
    }
    if (progressed != 0)
    {
      continue;
    }

    {
      uint64_t next_deadline = UINT64_MAX;
      if (transfer.sender_done == 0 && transfer.sender.timer_active != 0 &&
          transfer.sender.deadline_ms < next_deadline)
      {
        next_deadline = transfer.sender.deadline_ms;
      }
      if (transfer.receiver_done == 0 &&
          transfer.receiver.phase == RDT_RECEIVER_ACTIVE &&
          transfer.receiver.idle_deadline_ms < next_deadline)
      {
        next_deadline = transfer.receiver.idle_deadline_ms;
      }
      if (transfer.receiver_done == 0 &&
          transfer.receiver.phase == RDT_RECEIVER_LINGER &&
          transfer.receiver.linger_deadline_ms < next_deadline)
      {
        next_deadline = transfer.receiver.linger_deadline_ms;
      }
      if (next_deadline == UINT64_MAX)
      {
        return -1;
      }
      transfer.now_ms = next_deadline;
    }

    if (transfer.sender_done == 0 && transfer.sender.timer_active != 0 &&
        transfer.sender.deadline_ms <= transfer.now_ms)
    {
      sender_event.type = RDT_SENDER_EVENT_TIMER;
      sender_event.now_ms = transfer.now_ms;
      if (rdt_sender_step(&transfer.sender, &sender_event, &actions) != 0 ||
          drive_sender_actions(&transfer, &actions) != 0)
      {
        fprintf(stderr, "seed=%u sender timer action failed at %zu\n", seed,
                iteration);
        return -1;
      }
    }
    if (transfer.receiver_done == 0 &&
        ((transfer.receiver.phase == RDT_RECEIVER_ACTIVE &&
          transfer.receiver.idle_deadline_ms <= transfer.now_ms) ||
         (transfer.receiver.phase == RDT_RECEIVER_LINGER &&
          transfer.receiver.linger_deadline_ms <= transfer.now_ms)))
    {
      receiver_event.type = RDT_RECEIVER_EVENT_TIMER;
      receiver_event.now_ms = transfer.now_ms;
      if (rdt_receiver_step(&transfer.receiver, &receiver_event, &actions) !=
              0 ||
          drive_receiver_actions(&transfer, &actions) != 0)
      {
        fprintf(stderr, "seed=%u receiver timer action failed at %zu\n", seed,
                iteration);
        return -1;
      }
    }
    if (transfer.status != 0)
    {
      return -1;
    }
  }

  if (iteration == 100000U || transfer.status != 0 ||
      transfer.sender_done == 0 || transfer.receiver_done == 0 ||
      transfer.source_offset != source_length ||
      transfer.destination_length != source_length ||
      memcmp(source, transfer.destination, source_length) != 0)
  {
    fprintf(stderr, "seed=%u iterations=%zu status=%d sender=%d receiver=%d "
                    "sent=%zu received=%zu queued=%zu/%zu time=%llu\n",
            seed, iteration, transfer.status, transfer.sender_done,
            transfer.receiver_done, transfer.source_offset,
            transfer.destination_length, transfer.sender_to_receiver.count,
            transfer.receiver_to_sender.count,
            (unsigned long long) transfer.now_ms);
    return -1;
  }
  impairment_counts[0] = transfer.sender_to_receiver.dropped;
  impairment_counts[1] = transfer.sender_to_receiver.corrupted;
  impairment_counts[2] = transfer.sender_to_receiver.duplicated;
  impairment_counts[3] = transfer.receiver_to_sender.dropped;
  impairment_counts[4] = transfer.receiver_to_sender.corrupted;
  impairment_counts[5] = transfer.receiver_to_sender.duplicated;
  return 0;
}

static void test_state_machines_transfer_with_seeded_impairments(void)
{
  static const uint32_t seeds[] = {
    UINT32_C(1), UINT32_C(0x12345678), UINT32_C(0xdeadbeef),
    UINT32_C(0x31415926), UINT32_C(0xabcdef01)
  };
  unsigned int impairment_totals[6] = {0U};
  size_t i;

  for (i = 0U; i < sizeof(seeds) / sizeof(seeds[0]); ++i)
  {
    unsigned int counts[6];
    size_t count_index;

    TEST_ASSERT_EQUAL_INT(
        0, run_seeded_state_machine_transfer(seeds[i], counts));
    for (count_index = 0U; count_index < 6U; ++count_index)
    {
      impairment_totals[count_index] += counts[count_index];
    }
  }
  for (i = 0U; i < 6U; ++i)
  {
    TEST_ASSERT_TRUE(impairment_totals[i] > 0U);
  }
}

static void run_relay_response_case(int sender_role, unsigned int window_size,
                                    const char *response, int expected_status)
{
  struct rdt_options options = {
    .session = sender_role != 0 ? "test-rdt-send-err" : "test-rdt-recv-err",
    .window_size = window_size,
    .timeout_ms = 50U,
    .port = 0U,
    .loss = 0.0,
    .corrupt = 0.0,
    .duplicate = 0.0
  };
  char input_path[] = "/tmp/cs425-rdt-hello-XXXXXX";
  uint8_t datagram[128];
  unsigned short port = 0U;
  int relay_fd = make_relay_socket(&port);
  int input_fd = -1;
  int datagram_length;
  int status = 0;
  unsigned int attempt;
  pid_t worker;
  struct sockaddr_in peer;

  TEST_ASSERT_TRUE(relay_fd >= 0);
  options.port = port;
  if (sender_role != 0)
  {
    input_fd = mkstemp(input_path);
    TEST_ASSERT_TRUE(input_fd >= 0);
    (void) close(input_fd);
  }

  (void) fflush(NULL);
  worker = fork();
  TEST_ASSERT_TRUE(worker >= 0);
  if (worker == 0)
  {
    if (sender_role == 2)
    {
      struct rdt_io_context *io =
          rdt_io_open("127.0.0.1", port, input_path, &options, 1, NULL);
      if (io == NULL)
      {
        exit(1);
      }
      exit(rdt_io_close(io) == 0 ? 0 : 1);
    }
    if (sender_role != 0)
    {
      exit(rdt_send_file("127.0.0.1", port, input_path, &options));
    }
    exit(rdt_receive_file("127.0.0.1", port, "/tmp/unused-rdt-output",
                          &options));
  }

  for (attempt = 0U; attempt < (response == NULL ? 5U : 1U); ++attempt)
  {
    datagram_length = receive_relay_datagram(relay_fd, datagram,
                                             sizeof(datagram), &peer, 2000);
    TEST_ASSERT_TRUE(datagram_length > 0);
    if (response != NULL)
    {
      TEST_ASSERT_EQUAL_INT(0, send_relay_datagram(
                                   relay_fd, (const uint8_t *) response,
                                   strlen(response), &peer));
    }
  }
  TEST_ASSERT_EQUAL_INT(worker, waitpid(worker, &status, 0));
  TEST_ASSERT_TRUE(WIFEXITED(status));
  TEST_ASSERT_EQUAL_INT(expected_status, WEXITSTATUS(status));
  if (input_fd >= 0)
  {
    (void) unlink(input_path);
  }
  (void) close(relay_fd);
}
```
